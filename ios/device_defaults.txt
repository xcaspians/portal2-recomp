# Defaults the iOS app picks for each device, at first launch only and only
# for settings the player never changed (runtime/ios/device_defaults.c, read
# from the app bundle). "Use recommended" in the in-game menu applies them
# again. Tune the values here, then rebuild the app (tools/build_ios.py copies
# this file into it). Checked by tests/checks_device_defaults.c.
#
# tier <name> scale=<native|2|1.5|1> upscale=<1|0.75|0.5> fps=<30|40|60|120|auto>
#   scale:   Resolution (render scale over the screen in points; ×2 is the
#            measured default on an iPhone 15 Pro Max)
#   upscale: Upscaling (MetalFX): 0.75 renders at 75 % and upscales (Metal only)
#   fps:     Frames per second. 40 needs a 120 Hz (ProMotion) screen: on a
#            60 Hz screen it becomes 30. "auto" (120 on ProMotion) only once
#            a device holds ~90 fps in a busy chamber (round 1 rule): none yet.
#
# Conservative values: nothing below an A17 Pro has been measured. The A15
# (iPhone 13) has about half the GPU of the A17 Pro that holds 60 fps at ×2.
tier weak    scale=1   upscale=0.75 fps=30
tier a15     scale=1.5 upscale=0.75 fps=30
tier a15pro  scale=1.5 upscale=0.75 fps=40
tier a16     scale=2   upscale=0.75 fps=60
tier strong  scale=2   upscale=1    fps=60

# A model missing below: the safe middle.
unknown a16

# Low-memory guard: at or under this much memory (MB as iOS reports it: a
# 4 GB phone shows about 3,700), the game also starts with these console
# settings and these mtld3d memory caps, and logs its memory footprint at
# each level load (with the Log setting).
lowmem 4600
lowmem-args +mat_picmip 1
lowmem-metal memory.vramBudgetMB=512;memory.vbibRetentionCapMB=96;memory.pageboxPoolCapMB=32
